length_summary <- data_krill %>%

  filter(!is.na(length_mid)) %>%

  group_by(id_jar, species_code) %>%
  mutate(
    dens_species = sum(dens_number, na.rm = TRUE)) %>%
  ungroup() %>%

  
  filter(dens_species > 0) %>%

  group_by(year, species_code, length_mid) %>%
  summarise(
    frequency = sum(dens_number / dens_species, na.rm = TRUE),
    .groups = "drop" ) %>%

  
  
  arrange(year, species_code, length_mid) %>%

  group_by(year, species_code) %>%
  mutate(
    cumulative_frequency = cumsum(frequency),
    total_frequency = sum(frequency, na.rm = TRUE)) %>%

  
  summarise(
    median_length = length_mid[
      which(cumulative_frequency >= total_frequency * 0.50)[1]],
    
    q3_length = length_mid[
      which(cumulative_frequency >= total_frequency * 0.75)[1]],
    
    .groups = "drop")

plot_length_env <- function(data, species, metric, env_data, env_var) {

  length_df <- data %>%
    filter(species_code == species)

  plot_df <- length_df %>%
    left_join(
      env_data %>% select(year, all_of(env_var)),
      by = "year")

  
  # Labels
  metric_label <- ifelse(metric == "median_length", "Median", "Q3")

  env_label <- toupper(env_var)

  
  # Secondary axis transformation
  length_range <- range(plot_df[[metric]], na.rm = TRUE)
  env_range <- range(plot_df[[env_var]], na.rm = TRUE)

  scale_factor <- diff(length_range) / diff(env_range)
  offset <- length_range[1] - env_range[1] * scale_factor

  padding <- diff(length_range) * 0.10
  y_limits <- c(
  length_range[1] - padding,
  length_range[2] + padding)
  
  
 ggplot(plot_df, aes(x = factor(year))) +

  # Body length bars
  geom_col(
    aes(y = .data[[metric]]),
    fill = species_colors[species],
    width = 0.7,
    alpha = 0.8) +

   
  # Environmental index line
  geom_line(
    aes(y = .data[[env_var]] * scale_factor + offset, group = 1),
    color = env_colors[env_var],
    linewidth = 1,
    na.rm = TRUE) +

  geom_point(
    aes(y = .data[[env_var]] * scale_factor + offset),
    color = env_colors[env_var],
    size = 2,
    na.rm = TRUE) +

  scale_y_continuous(
    name = paste0(metric_label, " body length (mm)"),
    sec.axis = sec_axis(
      ~ (. - offset) / scale_factor,
      name = env_label
    )
  ) +
   
   
scale_x_discrete(drop = TRUE)  +
   
  labs(
    x = "Year",
    title =  paste0( "Annual " ,
      species, ": ", metric_label,
      " body length vs. ", env_label)) +

   
  theme_classic() +
  theme(
    plot.title = element_text(size = 16, face = "bold"),
    axis.title = element_text(size = 14),
    axis.text.x = element_text(angle = 45, hjust = 1),
    axis.text.y = element_text(size = 12),
    axis.title.y.left = element_text(color = "black"),
    axis.title.y.right = element_text(color = "black")
  )}

plot_length_scatter <- function(data, species, metric, env_var) {


plot_df <- data %>%
  filter(species_code == species) %>%
  filter(
    is.finite(.data[[env_var]]),
    is.finite(.data[[metric]])
  )

p_value <- cor.test(
  plot_df[[env_var]],
  plot_df[[metric]],
  method = "pearson"
)$p.value

p_color <- ifelse(p_value < 0.05, "forestgreen", "#a91b0d")

  # Labels
  metric_label <- ifelse(metric == "median_length", "Median", "Q3")
  env_label <- toupper(env_var)

  ggplot(plot_df,
         aes(x = .data[[env_var]], y = .data[[metric]])) +

    geom_point(
      color = species_colors[species],
      size = 2.5,
      alpha = 0.8,
      na.rm = TRUE
    ) +

    geom_smooth(
      method = "lm",
      formula = y ~ x,
      se = TRUE,
      color = env_colors[env_var],,
      fill = env_colors[env_var],,
      alpha = 0.2,
      na.rm = TRUE
    ) +
    
    # Regression equation
    stat_regline_equation(
      aes(label = after_stat(eq.label)),
      label.x.npc = "left",
      label.y.npc = 0.98
    ) +

# Pearson r
stat_cor(
  method = "pearson",
  aes(label = after_stat(r.label)),
  label.x.npc = "left",
  label.y.npc = 0.88,
  color = "black"
) +

# P-value
stat_cor(
  method = "pearson",
  aes(label = after_stat(p.label)),
  label.x.npc = "left",
  label.y.npc = 0.80,
  color = p_color
) +

    labs(
      x = env_label,
      y = paste0(metric_label, " body length (mm)"),
      title = paste0(
        species, ": ", metric_label,
        " body length vs. ", env_label
      )
    ) +

    theme_classic() +
    theme(
      plot.title = element_text(size = 16, face = "bold"),
      axis.title = element_text(size = 14),
      axis.text = element_text(size = 12)
    )
}

plot_length_combined <- function(data1, data2, species, metric,
                                 env_data, env_var,
                                 widths = c(8, 10)) {

  # Labels
  metric_label <- ifelse(metric == "median_length", "Median", "Q3")
  env_label <- toupper(env_var)

  # Shared body length limits
  y_limits <- range(
    c(
      data1[[metric]][data1$species_code == species],
      data2[[metric]][data2$species_code == species]
    ),
    na.rm = TRUE
  )

  # Left: cruise-level scatterplot
  p_left <- plot_length_scatter(
    data2, species, metric, env_var
  ) +
    labs(
      title = "Cruise-level",
      y = paste0(metric_label, " body length (mm)")
    ) +
    coord_cartesian(ylim = y_limits) +
    theme(
      plot.title = element_text(size = 13, face = "bold")
    )

  # Right: annual bar plot
  p_right <- plot_length_env(
    data1, species, metric, env_data, env_var
  ) +
    labs(
      title = "Annual"
    ) +
    coord_cartesian(ylim = y_limits) +
    theme(
      plot.title = element_text(size = 13, face = "bold"),
      axis.title.y.left = element_blank(),
      axis.text.y.left = element_blank()
    )

  # Combine
  p_left + p_right +
    plot_layout(widths = widths) +
    plot_annotation(
      title = paste0(
        species, ": ", metric_label,
        " body length vs. ", env_label
      ),
      theme = theme(
        plot.title = element_text(size = 17, face = "bold"))
      )
}

  lag_labels <- c("No lag", "1-month lag",
                  "2-month lag", "3-month lag")

  metric_label <- ifelse(metric == "median_length", "Median", "Q3")
  env_label <- toupper(index)

  # Prepare data
  plot_df <- data %>%
    filter(species_code == species) %>%
    select(all_of(c(metric, lag_vars))) %>%
    pivot_longer(
      cols = all_of(lag_vars),
      names_to = "lag",
      values_to = "env_value"
    ) %>%
    filter(
      is.finite(env_value),
      is.finite(.data[[metric]])
    )

  # Create one plot per lag
  plots <- lapply(seq_along(lag_vars), function(i) {

    lag_df <- plot_df %>%
      filter(lag == lag_vars[i])

    # Regression and correlation
    model <- lm(
      reformulate("env_value", response = metric),
      data = lag_df
    )

    intercept <- coef(model)[1]
    slope <- coef(model)[2]

    cor_result <- cor.test(
      lag_df$env_value,
      lag_df[[metric]],
      method = "pearson"
    )

    r <- unname(cor_result$estimate)
    p <- cor_result$p.value

    p_color <- ifelse(p < 0.05, "forestgreen", "#a91b0d")

    # Annotation labels
    eq_label <- paste0(
      "y = ", round(intercept, 2),
      ifelse(slope < 0, " - ", " + "),
      abs(round(slope, 2)), "x"
    )

    r_label <- paste0("r = ", round(r, 3))
    p_label <- paste0(
      "p = ", format.pval(p, digits = 2, eps = 0.001)
    )

    # Scatterplot
    p_scatter <- ggplot(
      lag_df,
      aes(x = env_value, y = .data[[metric]])
    ) +
      geom_point(
        color = species_colors[species],
        size = 2.5,
        alpha = 0.8
      ) +
      geom_smooth(
        method = "lm",
        formula = y ~ x,
        se = TRUE,
        color = env_colors[index],
        fill = env_colors[index],
        alpha = 0.2
      ) +
      labs(
        title = lag_labels[i],
        x = env_label,
        y = paste0(metric_label, " body length (mm)")
      ) +
      theme_classic() +
      theme(
        plot.title = element_text(size = 13, face = "bold"),
        axis.title = element_text(size = 12),
        axis.text = element_text(size = 11)
      )

    # Annotation panel
    p_stats <- ggplot() +
      annotate(
        "text", x = 0, y = 0.75,
        label = eq_label,
        hjust = 0, size = 4
      ) +
      annotate(
        "text", x = 0, y = 0.55,
        label = r_label,
        hjust = 0, size = 4
      ) +
      annotate(
        "text", x = 0, y = 0.35,
        label = p_label,
        hjust = 0, size = 4,
        color = p_color
      ) +
      xlim(0, 1) +
      ylim(0, 1) +
      theme_void()

    # Combine scatter + statistics
    p_scatter + p_stats +
      patchwork::plot_layout(widths = c(4, 1.5))
  })

  # Stack all four plots
  patchwork::wrap_plots(plots, ncol = 1) +
    patchwork::plot_annotation(
      title = paste0(
        species, ": ", metric_label,
        " body length vs. ", env_label
      ),
      theme = theme(
        plot.title = element_text(size = 16, face = "bold")
      )
    )
}
